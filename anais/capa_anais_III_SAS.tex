% Símbolo vetorial inspirado na marca fornecida:
% raios alaranjados, engrenagem azul e rede de nós com a sigla SAS.
\newcommand{\SASemblem}[1][1]{%
\begin{tikzpicture}[scale=#1,line cap=round,line join=round]
  % Raios solares
  \foreach \a in {0,15,...,345}{
    \begin{scope}[rotate=\a]
      \fill[SASOrange] (-0.065,1.42) rectangle (0.065,1.83);
    \end{scope}
  }
  % Corpo da engrenagem
  \fill[SASNavy] (0,0) circle (1.05);
  \foreach \a in {0,30,...,330}{
    \begin{scope}[rotate=\a]
      \fill[SASNavy] (-0.17,0.88) rectangle (0.17,1.18);
    \end{scope}
  }
  % Anel branco e contorno interno
  \fill[white] (0,0) circle (0.79);
  \draw[SASBlue,line width=2.2pt] (0,0) circle (0.79);
  \draw[SASOrange,line width=2.2pt] (0,-0.79) arc[start angle=-90,end angle=90,radius=0.79];

  % Rede de conexões
  \draw[SASNavy,line width=1.5pt]
    (0,0.48)--(-0.43,0.16)--(-0.43,-0.16)--(0,-0.49)--(0.43,-0.16)--(0.43,0.16)--cycle;
  \draw[SASNavy,line width=1.5pt] (0,0.48)--(0,-0.49);
  \draw[SASNavy,line width=1.5pt] (-0.43,0.16)--(0.43,0.16);
  \foreach \x/\y in {0/0.48,-0.43/0.16,0.43/0.16,-0.43/-0.16,0.43/-0.16,0/-0.49}{
    \fill[SASNavy] (\x,\y) circle (0.055);
  }
  \node[font=\sffamily\bfseries\fontsize{24}{24}\selectfont,text=SASNavy,
        fill=white,inner sep=1pt] at (0,0) {SAS};
\end{tikzpicture}%
}
